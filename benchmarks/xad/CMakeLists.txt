cmake_minimum_required(VERSION 3.18)
project(rust_aad_xad_replay LANGUAGES CXX)
if(NOT EXISTS "${XAD_SOURCE_DIR}/src/XAD/XAD.hpp")
    message(FATAL_ERROR "Set XAD_SOURCE_DIR to an XAD source checkout")
endif()
set(XAD_ENABLE_TESTS OFF CACHE BOOL "")
# Current Apple SDKs warn about the deployment target in XAD 2.2.0.
set(XAD_WARNINGS_PARANOID OFF CACHE BOOL "")
add_subdirectory("${XAD_SOURCE_DIR}" xad-build EXCLUDE_FROM_ALL)
add_executable(xad-affine affine.cpp)
target_compile_features(xad-affine PRIVATE cxx_std_20)
target_link_libraries(xad-affine PRIVATE xad)
if(MSVC)
    target_compile_options(xad-affine PRIVATE /fp:strict)
else()
    target_compile_options(xad-affine PRIVATE -ffp-contract=off)
endif()
